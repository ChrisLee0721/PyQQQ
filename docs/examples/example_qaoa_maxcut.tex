\subsection{数学推导}

QAOA MaxCut 优化：
\begin{equation}
C(z) = \sum_{(i,j) \in E} w_{ij} (1 - z_i z_j)
\end{equation}

\textbf{QAOA 电路}：
\begin{equation}
|\gamma, \beta\rangle = \prod_{l=1}^{p} e^{-i\beta_l B} e^{-i\gamma_l C} |+\rangle^{\otimes n}
\end{equation}
