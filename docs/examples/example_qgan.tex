\section{为什么需要？}

量子 GAN 是经典 GAN 的量子推广。

\subsection{经典局限}
\begin{itemize}
  \item 经典 GAN 训练不稳定
  \item 对于复杂数据，经典方法效率低
  \item 经典 GAN 容易陷入模式坍塌
\end{itemize}

\subsection{量子优势}
\begin{itemize}
  \item 量子 GAN 可以生成量子态和经典数据
  \item 训练更稳定
  \item 可以生成更复杂的数据
\end{itemize}

\subsection{实际应用}
\begin{itemize}
  \item 生成模型
  \item 数据增强

\subsection{数学推导}

\textbf{Step 1: 生成器}

生成器：
\begin{equation}
G(z) = U_G(\theta_G)|z\rangle
\end{equation}

\textbf{Step 2: 判别器}

判别器：
\begin{equation}
D(x) = \langle x|U_D^\dagger(\theta_D) M U_D(\theta_D)|x\rangle
\end{equation}

\textbf{Step 3: 损失函数}

生成器损失：
\begin{equation}
L_G = -\mathbb{E}[\log D(G(z))]
\end{equation}

判别器损失：
\begin{equation}
L_D = -\mathbb{E}[\log D(x)] - \mathbb{E}[\log(1 - D(G(z)))]
\end{equation}

\textbf{Step 4: 训练}

交替优化生成器和判别器：
\begin{equation}
\theta_G^* = \arg\min_{\theta_G} L_G
\end{equation}
\begin{equation}
\theta_D^* = \arg\min_{\theta_D} L_D
\end{equation}

\subsection{几何解释}

量子 GAN 在量子态空间中生成数据。生成器将噪声映射到数据分布，判别器区分真实数据和生成数据。

\section{代码详解}

\begin{verbatim}
from quonic.ml import qgan

# qgan(data, n_epochs, learning_rate)
# data: 训练数据
# n_epochs: 训练轮数
# learning_rate: 学习率

result = qgan(
    data=[[0, 1], [1, 0], [1, 1], [0, 0]],
    n_epochs=100,
    learning_rate=0.01
)

# result.generated: 生成的数据
# result.loss: 损失值
# result.generator_params: 生成器参数
# result.discriminator_params: 判别器参数
print(f'Generated: {result.generated}')
print(f'Loss: {result.loss}')
\end{verbatim}

\section{进阶用法}

\subsection{场景 1：量子态生成}

使用 QGAN 生成特定量子态：

\begin{verbatim}
from quonic.ml import qgan
import numpy as np

# 目标：生成 Bell 态 |00> + |11> 的样本
# 训练数据：Bell 态的测量结果
bell_data = [
    [0, 0],  # |00>
    [1, 1],  # |11>
    [0, 0],
    [1, 1],
    [0, 0],
    [1, 1],
]

# 训练 QGAN
result = qgan(data=bell_data, n_epochs=200)

# 生成新样本
generated = result.generate(n_samples=10)
print(f'Generated samples: {generated}')

# 分析生成样本的分布
prob_00 = sum(1 for s in generated if s == [0, 0]) / len(generated)
prob_11 = sum(1 for s in generated if s == [1, 1]) / len(generated)
prob_01 = sum(1 for s in generated if s == [0, 1]) / len(generated)
prob_10 = sum(1 for s in generated if s == [1, 0]) / len(generated)

print(f'P(|00>) = {prob_00:.2f}')
print(f'P(|11>) = {prob_11:.2f}')
print(f'P(|01>) = {prob_01:.2f}')
print(f'P(|10>) = {prob_10:.2f}')

# 分析：QGAN 学习了 Bell 态的分布
# 理想情况下，P(|00>) = P(|11>) = 0.5
\end{verbatim}

\subsection{场景 2：图像生成（二值图像）}

使用 QGAN 生成二值图像：

\begin{verbatim}
from quonic.ml import qgan
import numpy as np
import matplotlib.pyplot as plt

# 训练数据：简单的 4x4 二值图像
# 图案：对角线
diagonal_data = []
for _ in range(20):
    img = np.zeros((4, 4))
    for i in range(4):
        img[i, i] = 1
    diagonal_data.append(img.flatten().tolist())

# 训练 QGAN
result = qgan(data=diagonal_data, n_epochs=200)

# 生成新图像
generated_images = result.generate(n_samples=5)

# 可视化
fig, axes = plt.subplots(2, 5, figsize=(15, 6))

# 原始图像
for i in range(5):
    axes[0, i].imshow(np.array(diagonal_data[i]).reshape(4, 4), cmap='gray')
    axes[0, i].set_title('Original')
    axes[0, i].axis('off')

# 生成图像
for i in range(5):
    axes[1, i].imshow(np.array(generated_images[i]).reshape(4, 4), cmap='gray')
    axes[1, i].set_title('Generated')
    axes[1, i].axis('off')

plt.tight_layout()
plt.savefig('qgan_images.png')
plt.show()

# 分析：QGAN 学习了对角线图案的分布
# 生成的图像应该具有相似的对角线结构
\end{verbatim}

\subsection{场景 3：训练稳定性分析}

分析 QGAN 的训练稳定性：

\begin{verbatim}
from quonic.ml import qgan
import matplotlib.pyplot as plt

# 训练数据
data = [[0, 1], [1, 0], [1, 1], [0, 0]]

# 训练 QGAN 并记录损失
result = qgan(data=data, n_epochs=500, record_history=True)

# 绘制训练曲线
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(result.history['generator_loss'], label='Generator')
plt.plot(result.history['discriminator_loss'], label='Discriminator')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.title('QGAN Training Loss')
plt.legend()
plt.grid(True)

plt.subplot(1, 2, 2)
plt.plot(result.history['fidelity'], label='Fidelity')
plt.xlabel('Epoch')
plt.ylabel('Fidelity')
plt.title('Generated Data Fidelity')
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.savefig('qgan_training.png')
plt.show()

# 分析 QGAN 的训练稳定性：
# 1. 生成器和判别器的损失应该收敛
# 2. 生成数据的保真度应该增加
# 3. QGAN 比经典 GAN 更稳定（量子纠缠帮助）

# 与经典 GAN 对比
from quonic.ml import gan
classic_result = gan(data=data, n_epochs=500)
print(f'QGAN fidelity: {result.history["fidelity"][-1]:.4f}')
print(f'Classic GAN fidelity: {classic_result.history["fidelity"][-1]:.4f}')
\end{verbatim}
