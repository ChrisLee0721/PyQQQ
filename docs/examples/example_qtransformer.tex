\section{为什么需要？}

量子 Transformer 是经典 Transformer 的量子推广。

\subsection{经典局限}
\begin{itemize}
  \item 经典 Transformer 训练慢
  \item 对于长序列，经典方法效率低
  \item 经典 Transformer 受限于有限的注意力机制
\end{itemize}

\subsection{量子优势}
\begin{itemize}
  \item 量子 Transformer 可以处理长序列
  \item 训练速度更快
  \item 可以提取更复杂的特征
\end{itemize}

\subsection{实际应用}
\begin{itemize}
  \item 自然语言处理
  \item 序列建模
  \item 量子机器学习
\end{itemize}

\subsection{数学推导}

\textbf{Step 1: 量子注意力}

量子注意力：
\begin{equation}
\text{Attention}(Q,K,V) = \text{softmax}(QK^T/\sqrt{d_k})V
\end{equation}

\textbf{Step 2: 量子 Q、K、V}

量子 Q、K、V：
\begin{equation}
Q = U_Q(\theta_Q)|x\rangle, \quad K = U_K(\theta_K)|x\rangle, \quad V = U_V(\theta_V)|x\rangle
\end{equation}

\textbf{Step 3: 量子内积}

量子内积：
\begin{equation}
\langle Q_i|K_j\rangle = \langle 0|U_Q^\dagger U_K|0\rangle
\end{equation}

\textbf{Step 4: 量子 softmax}

量子 softmax：
\begin{equation}
\text{softmax}(x_i) = \frac{e^{x_i}}{\sum_j e^{x_j}}
\end{equation}

\textbf{Step 5: 量子输出}

量子输出：
\begin{equation}
\text{Output} = \sum_j \alpha_j V_j
\end{equation}

\subsection{几何解释}

量子 Transformer 在量子态空间中处理序列数据。量子注意力计算序列中不同位置之间的关联。

\section{代码详解}

\begin{verbatim}
from quonic.ml import qtransformer

# qtransformer(data, n_heads, n_layers, n_epochs)
# data: 训练数据
# n_heads: 注意力头数
# n_layers: 层数
# n_epochs: 训练轮数

result = qtransformer(
    data=[[0, 1], [1, 0], [1, 1], [0, 0]],
    n_heads=2,
    n_layers=2,
    n_epochs=100
)

# result.output: 输出
# result.accuracy: 准确率
# result.loss: 损失值
# result.params: 参数
print(f'Output: {result.output}')
print(f'Accuracy: {result.accuracy}')
\end{verbatim}
