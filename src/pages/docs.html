Quick Start

PythonCopy
from quonic import qgate, qshow
from quonic.gates import H, CX

# Bell state
qgate(H, 0)
qgate(CX, 0, 1)
qshow()

qgate()

qgate(gate, *qubits, **params)
Apply a quantum gate to specified qubits.
gateGate|strGate object or name string
*qubitsintQubit indices
**paramsfloatGate parameters (for parametric gates)
Examples
qgate(H, 0)           # Hadamard on qubit 0
qgate(CX, 0, 1)       # CNOT: control=0, target=1
qgate(Ry, 0, 0.5)     # Ry(0.5) on qubit 0
qgate("h", 0)          # Gate by name

qshow()

qshow(backend='auto', shots=1024, noise=None)
Run the current circuit and display results.
backendstrBackend name ('auto', 'qiskit', 'cirq', 'native', etc.)
shotsintNumber of measurement shots
noisefloat|NoiseModelNoise level or model

Grover Search

Grover's Algorithm
from quonic.algorithms import grover

# Search for |11> in 2-qubit space
result = grover("11", 2, shots=1024)
print(result.counts)  # {'11': ~1000}

VQE

Variational Quantum Eigensolver
from quonic.algorithms import vqe

hamiltonian = [(1.0, "ZZ"), (0.5, "XI"), (0.5, "IX")]
result = vqe(hamiltonian, n_qubits=2, maxiter=200)
print(f"Energy: {result.value}")

ZNE Error Mitigation

result = zne(circuit, noise=0.05, extrapolation="linear")

Error Correction

Quantum Error Correction
from quonic.qec import BitFlipCode, qec_round_trip

result = qec_round_trip(code="bit_flip", error_rate=0.01)
print(f"Logical error: {result.logical_error_rate:.4f}")
